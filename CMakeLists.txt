cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SUN_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(SunKeyboardTools VERSION ${SUN_VERSION} LANGUAGES CXX RC)
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
configure_file(src/version.rc.in generated/version.rc @ONLY)
include_directories("${CMAKE_BINARY_DIR}/generated")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Sun Keyboard Tools currently targets Windows.")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc /sdl /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /CETCOMPAT /DEPENDENTLOADFLAG:0x800)
add_library(sun-core src/core.cpp)
target_include_directories(sun-core PUBLIC src)
add_library(sun-windows src/windows.cpp src/security.cpp)
target_link_libraries(sun-windows PUBLIC taskschd comsuppw shell32 ole32 oleaut32 advapi32 userenv)
add_library(sun-ipc src/ipc.cpp)
target_link_libraries(sun-ipc PUBLIC sun-windows sun-core)
add_library(sun-backend src/backend.cpp)
target_link_libraries(sun-backend PUBLIC sun-core sun-ipc user32)
add_executable(sun-keyboard-capture WIN32 src/broker.cpp src/capture.cpp "${CMAKE_BINARY_DIR}/generated/version.rc")
target_link_libraries(sun-keyboard-capture PRIVATE sun-ipc)
target_link_options(sun-keyboard-capture PRIVATE /MANIFESTUAC:level='requireAdministrator')
add_executable(sun-keyboard-tools WIN32 src/app.cpp src/app.manifest src/app.rc "${CMAKE_BINARY_DIR}/generated/version.rc")
target_link_libraries(sun-keyboard-tools PRIVATE sun-backend comctl32)
target_link_options(sun-keyboard-tools PRIVATE /MANIFESTUAC:NO)
add_custom_target(sun-runtime-files
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/profiles" "$<TARGET_FILE_DIR:sun-keyboard-tools>/profiles"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/docs/README.html" "$<TARGET_FILE_DIR:sun-keyboard-tools>/README.html"
    COMMENT "Refresh keyboard profiles and offline help")
add_dependencies(sun-keyboard-tools sun-runtime-files)
include(CTest)
if(BUILD_TESTING)
    add_executable(sun-core-tests tests/core_tests.cpp)
    target_link_libraries(sun-core-tests PRIVATE sun-core)
    add_test(NAME core COMMAND sun-core-tests "${CMAKE_SOURCE_DIR}/profiles")
    add_executable(sun-maintenance-tests tests/maintenance_tests.cpp)
    target_include_directories(sun-maintenance-tests PRIVATE src)
    add_test(NAME maintenance COMMAND sun-maintenance-tests)
    add_executable(sun-security-tests tests/security_tests.cpp)
    target_link_libraries(sun-security-tests PRIVATE sun-ipc)
    add_test(NAME security COMMAND sun-security-tests)
    add_executable(sun-windows-boundary-tests tests/windows_boundary.cpp)
    target_link_libraries(sun-windows-boundary-tests PRIVATE sun-ipc)
endif()
add_executable(sun-hid-probe tools/hid_probe.cpp)
target_link_libraries(sun-hid-probe PRIVATE hid setupapi advapi32)
add_executable(sun-key-monitor tools/key_monitor.cpp)
target_link_libraries(sun-key-monitor PRIVATE sun-backend)

option(SUN_BUILD_SETUP "Build the offline installer with the signed USBPcap package and its source" OFF)
set(SUN_DEPS_DIR "${CMAKE_SOURCE_DIR}/.local/dependencies" CACHE PATH "Verified USBPcap release files")
if(SUN_BUILD_SETUP)
    file(SHA256 "${SUN_DEPS_DIR}/USBPcapSetup-1.5.4.0.exe" DRIVER_SHA)
    file(SHA256 "${SUN_DEPS_DIR}/usbpcap-src-incl-pdb-1.5.4.0.7z" SOURCE_SHA)
    if(NOT DRIVER_SHA STREQUAL "87a7edf9bbbcf07b5f4373d9a192a6770d2ff3add7aa1e276e82e38582ccb622" OR
       NOT SOURCE_SHA STREQUAL "3a66cf03425f5860ff8acd64d11cc0bb50669c7f449754f47788011259a5b15e")
        message(FATAL_ERROR "USBPcap dependency hash mismatch. Fetch the exact official 1.5.4.0 release.")
    endif()
    configure_file(src/setup.rc.in "${CMAKE_BINARY_DIR}/setup.rc.in" @ONLY)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/setup-$<CONFIG>.rc" INPUT "${CMAKE_BINARY_DIR}/setup.rc.in")
    add_executable(SunKeyboardTools-Maintenance WIN32 src/setup.cpp "${CMAKE_BINARY_DIR}/setup-$<CONFIG>.rc" "${CMAKE_BINARY_DIR}/generated/version.rc")
    target_link_libraries(SunKeyboardTools-Maintenance PRIVATE sun-ipc comctl32 wintrust bcrypt)
    add_dependencies(SunKeyboardTools-Maintenance sun-keyboard-tools sun-keyboard-capture)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/maintenance-$<CONFIG>.rc" CONTENT "#include <windows.h>\n110 RCDATA \"$<TARGET_FILE:SunKeyboardTools-Maintenance>\"\n")
    add_executable(SunKeyboardTools-Setup WIN32 src/setup.cpp "${CMAKE_BINARY_DIR}/setup-$<CONFIG>.rc" "${CMAKE_BINARY_DIR}/maintenance-$<CONFIG>.rc" "${CMAKE_BINARY_DIR}/generated/version.rc")
    target_include_directories(SunKeyboardTools-Setup PRIVATE src)
    target_link_libraries(SunKeyboardTools-Setup PRIVATE sun-ipc comctl32 wintrust bcrypt)
    add_dependencies(SunKeyboardTools-Setup SunKeyboardTools-Maintenance)
endif()
