cmake_minimum_required(VERSION 3.24)
project(SunKeyboardTools VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Sun Keyboard Tools currently targets Windows.")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_library(sun-core src/core.cpp)
target_include_directories(sun-core PUBLIC src)
add_library(sun-windows src/windows.cpp)
target_link_libraries(sun-windows PUBLIC taskschd comsuppw shell32 ole32 oleaut32 advapi32)
add_library(sun-backend src/backend.cpp)
target_link_libraries(sun-backend PUBLIC sun-core sun-windows user32)
add_executable(sun-keyboard-tools WIN32 src/app.cpp src/app.manifest src/app.rc)
target_link_libraries(sun-keyboard-tools PRIVATE sun-backend comctl32)
target_link_options(sun-keyboard-tools PRIVATE /MANIFESTUAC:NO)
add_custom_target(sun-runtime-files
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/profiles" "$<TARGET_FILE_DIR:sun-keyboard-tools>/profiles"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/docs/README.html" "$<TARGET_FILE_DIR:sun-keyboard-tools>/README.html"
    COMMENT "Refresh keyboard profiles and offline help")
add_dependencies(sun-keyboard-tools sun-runtime-files)
include(CTest)
if(BUILD_TESTING)
    add_executable(sun-core-tests tests/core_tests.cpp)
    target_link_libraries(sun-core-tests PRIVATE sun-core)
    add_test(NAME core COMMAND sun-core-tests "${CMAKE_SOURCE_DIR}/profiles")
endif()
add_executable(sun-hid-probe tools/hid_probe.cpp)
target_link_libraries(sun-hid-probe PRIVATE hid setupapi advapi32)
add_executable(sun-key-monitor tools/key_monitor.cpp)
target_link_libraries(sun-key-monitor PRIVATE sun-backend)

option(SUN_BUILD_SETUP "Build the offline installer with the signed USBPcap package and its source" OFF)
set(SUN_DEPS_DIR "${CMAKE_SOURCE_DIR}/.local/dependencies" CACHE PATH "Verified USBPcap release files")
if(SUN_BUILD_SETUP)
    file(SHA256 "${SUN_DEPS_DIR}/USBPcapSetup-1.5.4.0.exe" DRIVER_SHA)
    file(SHA256 "${SUN_DEPS_DIR}/usbpcap-src-incl-pdb-1.5.4.0.7z" SOURCE_SHA)
    if(NOT DRIVER_SHA STREQUAL "87a7edf9bbbcf07b5f4373d9a192a6770d2ff3add7aa1e276e82e38582ccb622" OR
       NOT SOURCE_SHA STREQUAL "3a66cf03425f5860ff8acd64d11cc0bb50669c7f449754f47788011259a5b15e")
        message(FATAL_ERROR "USBPcap dependency hash mismatch. Fetch the exact official 1.5.4.0 release.")
    endif()
    configure_file(src/setup.rc.in "${CMAKE_BINARY_DIR}/setup.rc.in" @ONLY)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/setup-$<CONFIG>.rc" INPUT "${CMAKE_BINARY_DIR}/setup.rc.in")
    add_executable(SunKeyboardTools-Setup WIN32 src/setup.cpp "${CMAKE_BINARY_DIR}/setup-$<CONFIG>.rc")
    target_include_directories(SunKeyboardTools-Setup PRIVATE src)
    target_link_libraries(SunKeyboardTools-Setup PRIVATE sun-windows comctl32 wintrust)
    add_dependencies(SunKeyboardTools-Setup sun-keyboard-tools)
endif()
